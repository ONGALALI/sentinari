"""Sentinari Telemetry Extractor & Normalization Pipeline

Safely processes existing tables inside data/vocationalllm.db and
transforms text content into the canonical Article V structure.
"""

import os
import sqlite3
import json
import uuid
from datetime import datetime, timezone
from pathlib import Path

# Resolve workspace boundaries
ROOT = Path(__file__).resolve().parents[1]
DB_PATH = ROOT / "data" / "vocationalllm.db"
OUTPUT_TELEMETRY_DIR = ROOT / "data" / "telemetry_vault"
OUTPUT_TELEMETRY_DIR.mkdir(parents=True, exist_ok=True)

def initialize_telemetry_table(conn: sqlite3.Connection):
    """Establishes an immutable, structured table for Article V knowledge maps."""
    conn.execute("""
        CREATE TABLE IF NOT EXISTS article_v_knowledge (
            observation_id TEXT PRIMARY KEY,
            source_table TEXT NOT NULL,
            source_rowid INTEGER NOT NULL,
            timestamp TEXT NOT NULL,
            scientific_field TEXT,
            falsifiable_claims TEXT NOT NULL DEFAULT '[]',
            numerical_metrics TEXT NOT NULL DEFAULT '[]',
            epistemic_confidence REAL NOT NULL DEFAULT 1.0,
            provenance_json TEXT NOT NULL DEFAULT '{}'
        )
    """)
    conn.commit()

def extract_and_map_telemetry():
    if not DB_PATH.exists():
        raise FileNotFoundError(f"Source database not found at {DB_PATH}")

    conn = sqlite3.connect(str(DB_PATH))
    conn.row_factory = sqlite3.Row
    initialize_telemetry_table(conn)
    cur = conn.cursor()

    # Step 1: Detect candidate columns exactly like your RAG indexer
    from .RAG_Indexer import scan_text_columns
    candidates = scan_text_columns(conn)
    
    if not candidates:
        print("[-] No valid source columns identified for telemetry capture.")
        return

    processed_count = 0

    # Step 2: Iterate through your existing tables and extract captured texts
    for table, col in candidates:
        # Prevent parsing our own newly created telemetry table
        if table == "article_v_knowledge":
            continue

        print(f"[*] Extracting telemetry signatures from {table}.{col}...")
        
        try:
            cur.execute(f"SELECT rowid, {col} FROM {table} WHERE {col} IS NOT NULL")
            rows = cur.fetchall()
        except Exception as e:
            print(f"[-] Read failure on {table}.{col}: {e}")
            continue

        for row in rows:
            rowid = row["rowid"]
            raw_text = row[col]

            if not raw_text or len(str(raw_text).strip()) < 40:
                continue

            # Check if we have already mapped this specific database record
            exists = conn.execute(
                "SELECT 1 FROM article_v_knowledge WHERE source_table=? AND source_rowid=?",
                (table, rowid)
            ).fetchone()
            if exists:
                continue

            # Convert binary components if they exist
            if isinstance(raw_text, bytes):
                raw_text = raw_text.decode('utf-8', errors='ignore')

            # Step 3: Parse and enforce Article V structural properties
            observation_id = f"telemetry:local:{table}:{rowid}:{uuid.uuid4().hex[:8]}"
            timestamp_iso = datetime.now(timezone.utc).isoformat(timespec="seconds")
            
            # Simple heuristic rules to infer cross-functional scientific field focus
            inferred_field = "general-science"
            text_lower = raw_text.lower()
            if any(x in text_lower for x in ("gene", "covid", "virus", "protein", "receptor")):
                inferred_field = "life-and-biological-systems"
            elif any(x in text_lower for x in ("temp", "thermo", "velocity", "quantum")):
                inferred_field = "foundations-of-reality"
            elif any(x in text_lower for x in ("algorithm", "sqlite", "matrix", "network")):
                inferred_field = "computer-and-information-sciences"

            # Parse structural metrics out of raw data strings
            # For a deep production pass, patch a fast model endpoint here
            claims = [line.strip() for line in raw_text.split('.') if len(line) > 30][:3]
            
            provenance = {
                "origin_database": "vocationalllm.db",
                "origin_table": table,
                "origin_column": col,
                "origin_rowid": rowid,
                "ingest_timestamp": timestamp_iso
            }

            # Step 4: Persist formatted telemetry directly back into your localized schema
            try:
                conn.execute("""
                    INSERT INTO article_v_knowledge (
                        observation_id, source_table, source_rowid, timestamp, 
                        scientific_field, falsifiable_claims, numerical_metrics, 
                        epistemic_confidence, provenance_json
                    ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
                """, (
                    observation_id, table, rowid, timestamp_iso, inferred_field,
                    json.dumps(claims, ensure_ascii=False),
                    json.dumps({"text_length_chars": len(raw_text)}),
                    0.90, # Base epistemic rating
                    json.dumps(provenance)
                ))
                processed_count += 1
            except Exception as write_err:
                print(f"[-] Insertion failure for observation {observation_id}: {write_err}")

        conn.commit()
    
    print(f"[+] Telemetry pipeline sync complete. Structured {processed_count} new entries.")
    conn.close()

if __name__ == "__main__":
    extract_and_map_telemetry()
